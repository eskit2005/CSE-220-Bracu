    public static Node linkNodes(int[][] edges) {
        Node head = null;
        for (int i = edges.length - 1; i >= 0; i--) {
            head = new Node(new Edge(edges[i][0], edges[i][1]), head);
        }
        return head;
    }

   
    public static String maxDegreeList(Node[] adjList) {
        int max = 0;
        int vertexWithMaxDegree = -1;

        for (int i = 0; i < adjList.length; i++) {
            int count = 0;
            Node current = adjList[i];
            while (current != null) {
                count++;
                current = current.next;
            }
            if (count > max) {
                max = count;
                vertexWithMaxDegree = i;
            }
        }

        return "Adjency list's max vertex " + vertexWithMaxDegree + " and its degree is " + max;
    }

   
    public static String maxDegreeMatrix(int[][] matrix) {
        int max = 0;
        int vertexWithMaxDegree = -1;

        for (int i = 0; i < matrix.length; i++) {
            int count = 0;
            for (int j = 0; j < matrix[i].length; j++) {
                if (matrix[i][j] == 1) {
                    count++;
                }
            }
            if (count > max) {
                max = count;
                vertexWithMaxDegree = i;
            }
        }

        return "Adjency matrix's max vertex " + vertexWithMaxDegree + " and its degree is " + max;
    }

    public static String maxWeightedVertexMatrix(int[][] matrix) {

                total += matrix[i][j];

            if (total > maxSum) {

            }

                maxSum = total;
                heavyVertex = i;
            }
        }

    public static String getMaxDegreeFromMatrix(int[][] matrix) {
        int maxDegree = 0, node = -1;

        for (int i = 0; i < matrix.length; i++) {
            int degree = 0;
            for (int j = 0; j < matrix[i].length; j++) {
                if (matrix[i][j] > 0) degree++;
            }
            if (degree > maxDegree) {
                maxDegree = degree;
                node = i;
            }
        }
        return "Node " + node + ", Degree: " + maxDegree;
    }


    public static String getMaxWeightFromMatrix(int[][] matrix) {
        int maxWeight = 0, node = -1;

        for (int i = 0; i < matrix.length; i++) {

            for (int j = 0; j < matrix[i].length; j++) {
                sum += matrix[i][j];
            }
            if (sum > maxWeight) {
                maxWeight = sum;
                node = i;
            }
        }

    public static void connectMatrix(int[][] matrix, int from, int to, int weight) {
        matrix[from][to] = weight;

    public static int[][] convertMatrixToUndirected(int[][] matrix) {
        int n = matrix.length;
        int[][] undirected = new int[n][n];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (matrix[i][j] > 0) {
                    undirected[i][j] = matrix[i][j];
                    undirected[j][i] = matrix[i][j];
                }
            }
        }
        return undirected;
    }

    public static Node[] convertListToUndirected(Node[] directedList) {
        int n = directedList.length;
        Node[] undirected = new Node[n];
        boolean[][] visited = new boolean[n][n];

        for (int i = 0; i < n; i++) {
            Node current = directedList[i];
            while (current != null) {
                int from = current.edge.from;
                int to = current.edge.to;
                int weight = current.edge.weight;

                if (!visited[from][to] && !visited[to][from]) {
                    Node n1 = new Node(new Edge(from, to, weight));
                    n1.next = undirected[from];
                    undirected[from] = n1;

                    Node n2 = new Node(new Edge(to, from, weight));
                    n2.next = undirected[to];
                    undirected[to] = n2;

                    visited[from][to] = true;
                    visited[to][from] = true;
                }
                current = current.next;
            }
        }

        return undirected;
    }


    public static void addToList(Node[] list, int from, int to, int weight) {
        Node newNode = new Node(new Edge(from, to, weight));
        newNode.next = list[from];
        list[from] = newNode;
    }
